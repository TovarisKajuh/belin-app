// VERIFICATION KIT (scratchpad only). Renders Wave 6 documents OFFLINE from
// representative Slovenian data: no database, no storage, no session. Each
// section imports its template lazily, so sections for documents that do not
// exist yet are simply not requested.
//
//   cd C:\DevEnv\belin-app            (or the lane-b worktree)
//   set NODE_PATH=%CD%\node_modules    (PowerShell: $env:NODE_PATH="$PWD\node_modules")
//   npx tsx --tsconfig tsconfig.scripts.json <scratch>/kit/render-offline.tsx <out-dir> day abnahme po regie ...
//
// Sections: day completion abnahme po regie invoice co poster situacija ovira
import { mkdirSync, readFileSync, writeFileSync } from "node:fs";

const [outDir, ...sections] = process.argv.slice(2);
if (!outDir || sections.length === 0) {
  console.error("usage: render-offline.tsx <out-dir> <section...>");
  process.exit(2);
}
mkdirSync(outDir, { recursive: true });

async function photo(name: string, caption: string) {
  const { imageSize } = await import("@/lib/pdf/image-size");
  const bytes = readFileSync(`${SITE}/${name}`);
  const size = imageSize(bytes)!;
  return { bytes, width: size.width, height: size.height, caption };
}

async function save(name: string, element: Parameters<(typeof import("@/lib/pdf/theme"))["renderDocument"]>[0]) {
  const { renderDocument } = await import("@/lib/pdf/theme");
  const buffer = await renderDocument(element);
  writeFileSync(`${outDir}/${name}.pdf`, buffer);
  console.log(`${name}.pdf ${buffer.length} bytes`);
}

async function main(): Promise<void> {
  for (const section of sections) {
    if (!run[section]) {
      console.error(`unknown section ${section}`);
      process.exit(2);
    }
    await run[section]();
  }
}
