// VERIFICATION KIT (scratchpad only). Decodes the QR code in a PNG.
// Needs: npm install --prefix <scratch>/kit jsqr@1.4.0 pngjs@7   (scratch, never the repo)
// node <scratch>/kit/decode.mjs <page.png>
import { readFileSync } from "node:fs";
import { PNG } from "pngjs";
import jsQR from "jsqr";
const png = PNG.sync.read(readFileSync(process.argv[2]));
const r = jsQR(new Uint8ClampedArray(png.data), png.width, png.height);
console.log(r ? `QR: ${r.data}` : "QR: NOT FOUND");
