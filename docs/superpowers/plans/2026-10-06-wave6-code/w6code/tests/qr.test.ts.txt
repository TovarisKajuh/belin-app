import { create } from "qrcode";
import { describe, expect, it } from "vitest";
import { qrSvg, qrViewBox } from "@/lib/qr";

// The one QR encoder for paper and screen draws exactly what the encoder decided.

describe("qrSvg", () => {
  const url = "https://getbelin.com/sl/p/AbCdEfGhIjKlMnOpQrStUvWxYz0123456789abcdefg";

  it("draws exactly the modules the encoder set", () => {
    const qr = qrSvg(url, "Q");
    const { modules } = create(url, { errorCorrectionLevel: "Q" });
    let dark = 0;
    for (let r = 0; r < modules.size; r++) for (let c = 0; c < modules.size; c++) dark += modules.get(r, c) ? 1 : 0;
    // Every run is "M<col> <row>h<run>v1h-<run>z": the runs add up to the dark modules.
    const drawn = [...qr.path.matchAll(/h(\d+)v1/g)].reduce((sum, m) => sum + Number(m[1]), 0);
    expect(qr.size).toBe(modules.size);
    expect(drawn).toBe(dark);
  });

  it("starts with the top-left finder pattern and leaves a 4 module quiet zone", () => {
    const qr = qrSvg(url);
    expect(qr.path.startsWith("M0 0h7v1h-7z")).toBe(true);
    expect(qrViewBox(qr)).toBe(`-4 -4 ${qr.size + 8} ${qr.size + 8}`);
  });
});
