// Task 6.2 anchor edits, applied after 6.1b. Run from the worktree root.
const fs = require("fs");
function edit(file, pairs) {
  let s = fs.readFileSync(file, "utf8");
  for (const [a, b] of pairs) {
    if (!s.includes(a)) throw new Error(`${file}: miss ${a.slice(0, 90)}`);
    s = s.split(a).join(b);
  }
  fs.writeFileSync(file, s);
  console.log("edited", file);
}

/** At most \`max\` characters, cut at a word boundary and marked when cut. */
function clip(text: string, max: number): string {
  if (text.length <= max) return text;
  return \`\${text.slice(0, max).replace(/\\s+\\S*$/, "")} ...\`;
}

// final-report.ts: the old day loop and its date formatters go; one formatter replaces them.
{
  const file = "lib/data/final-report.ts";
  let s = fs.readFileSync(file, "utf8");
  const a = s.indexOf("  const dayRefs = buildDayReports(");
  const b = s.indexOf("  const now = new Date();");
  if (a < 0 || b < 0) throw new Error("final-report: day loop anchors missing");
  s = s.slice(0, a) + "  const dateFmt = (iso: string) => docDate(iso, locale, project.country);\n\n" + s.slice(b);
  const p = s.indexOf("  const period =");
  const q = s.indexOf("  const strings = completionStrings(locale);");
  if (p < 0 || q < 0) throw new Error("final-report: period anchors missing");
  s =
    s.slice(0, p) +
    "  const period =\n    dayRefs.length > 0 ? `${dateFmt(dayRefs[0].dateIso)} - ${dateFmt(dayRefs[dayRefs.length - 1].dateIso)}` : null;\n\n" +
    s.slice(q);
  const c = s.indexOf("/**\n * Photo bytes for one day.");
  if (c > 0) s = s.slice(0, c).trimEnd() + "\n";
  fs.writeFileSync(file, s);
  console.log("restructured", file);
}
