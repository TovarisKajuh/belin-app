const fs = require("fs");
function edit(file, pairs) {
  let s = fs.readFileSync(file, "utf8");
  for (const [a, b] of pairs) {
    if (!s.includes(a)) throw new Error(`${file}: miss ${a.slice(0, 90)}`);
    s = s.split(a).join(b);
  }
  fs.writeFileSync(file, s);
  console.log("edited", file);
}
